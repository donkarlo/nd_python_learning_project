\chapter{Virtual environment}


\section{Virtual environment}
To go to the virtual environment

\begin{minted}{bash}
        source ~/phd-venv/bin/activate
\end{minted}

and to leave it

\begin{minted}{bash}
        deactivate
\end{minted}

\paragraph{Listing installed packages}
\begin{minted}{bash}
            pip list --editable
\end{minted}
