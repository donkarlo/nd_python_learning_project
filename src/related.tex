    \subsection{Lists}
        \begin{minted}{yaml}
            homogeneity:
              - name: "homogeneous"
                description: "All robots have the same model and sensors"
              - name: "heterogeneous"
                description: "Robots have different models or sensor setups"
        \end{minted}
        and then in python you can call
        \begin{minted}{python}
            print(data["homogeneity"][0]["name"])         # homogeneous
            print(data["homogeneity"][0]["description"])  # All robots have the same model and sensors
            
            print(data["homogeneity"][1]["name"])         # heterogeneous
            print(data["homogeneity"][1]["description"])  # Robots have different models or sensor setups
        \end{minted}

\section{Pycharm}
    \subsection{settings}
        \subsubsection{Editor}
            \paragraph{Live templates}
                for example if you want that whenever you type ``mnt`` in .tex files you see a suggestion for
                \begin{minted}{latex}
                    \\begin\{minted\}\{python\}
                        s
                    \\end\{minted\}
                \end{minted}
                just go to \textit{settings>editor>live templates>}
                    and then from the right panel find latex and add ``mnt`` with the code you want. Do not forget to mark reformat and assign \textit{application} to it from low right panel
