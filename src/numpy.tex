\section{Shape and dimension}
    Shape is a tuple of dimension 2. shape can be (10,) or (10,1) and this two are different.
    \paragraph{reshaping}
        Reshaping of unknow lenth from (n,) to (1,n)
        \begin{minted}{python}
            a = np.array(range(9))
            a = a.reshape(1,-1)
        \end{minted}
        or to a column vector
        \begin{minted}{python}
                a = np.array(range(9))
                a = a.reshape(-1,1)
        \end{minted}
    \paragraph{ndim}
        \begin{minted}{python}
            a1 = np.array([1,3,5])
            print(a1.ndim) # prints 1
            
            a2 = np.array([[1,3,5]])
            print(a2.ndim) # prints 2
        \end{minted}
