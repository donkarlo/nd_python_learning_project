\section{Operators}
    \paragraph{**} unpacks a dictioonary into key values
        \begin{minted}{python}
            def f(a, b):
                ...
            d = {"a": 1, "b": 2}
            
            The foloowing two are both the same
            f(a=1, b=2)
            f(**d)
        \end{minted}
